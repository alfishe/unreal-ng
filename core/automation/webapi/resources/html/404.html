Quick Start

# List emulators
curl http://localhost:8090/api/v1/emulator

# Create and start emulator
curl -X POST http://localhost:8090/api/v1/emulator/start \
  -H "Content-Type: application/json" \
  -d '{"model": "128K"}'

# Get registers
curl http://localhost:8090/api/v1/emulator/{id}/registers

# Add a label
curl -X POST http://localhost:8090/api/v1/emulator/{id}/labels \
  -H "Content-Type: application/json" \
  -d '{"name": "main", "address": 32768, "type": "code"}'
