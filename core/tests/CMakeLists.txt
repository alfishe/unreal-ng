cmake_minimum_required(VERSION 3.16.0)

message(">> core/tests CMakeLists.txt")

# Project named 'core-tests' on C++
project(core-tests LANGUAGES CXX)

# C++ standard - MUST match the rest of the project (C++20) to avoid ABI mismatch
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Set output directories to match root CMakeLists.txt
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Root directory
get_filename_component(ROOT_DIR ../../ ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})

# Include CMake helper functions file
include(${ROOT_DIR}/cmake/func.cmake)

# Recording support (audio/video capture)
option(ENABLE_RECORDING "Enable audio/video recording support" ON)

# Set paths
set(CORE_DIR ${ROOT_DIR}/core/src)
set(TESTS_DIR ${ROOT_DIR}/core/tests)
set(GOOGLETEST_DIR ${ROOT_DIR}/lib/googletest)
message("CORE_DIR:${CORE_DIR}")
message("TESTS_DIR:${TESTS_DIR}")
message("GOOGLETEST_DIR:${GOOGLETEST_DIR}")

# Set include files search paths
include_directories(
    ${TESTS_DIR}
    ${CORE_DIR}
    ${GOOGLETEST_DIR}/googletest                                  # To serve src/gtest-internal-inl.h include
    ${GOOGLETEST_DIR}/googletest/include
)

# Add recording library include path if enabled
if (ENABLE_RECORDING)
    include_directories(${ROOT_DIR}/core/recording/src)
endif()

# MinGW CMake doesn't build gtest - make it explicitly
if (WIN32 AND NOT MSVC)
    set (GOOGLETEST_FILES
        ${GOOGLETEST_DIR}/googletest/src/gtest.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-assertion-result.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-death-test.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-filepath.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-matchers.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-port.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-printers.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-test-part.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-typed-test.cc
        ${GOOGLETEST_DIR}/googletest/src/gtest-internal-inl.h
    )
else (WIN32 AND NOT MSVC)
    # Compile GoogleTest to <binary>/googletest subfolder (skip if already defined)
    if (NOT TARGET gtest)
        add_subdirectory(${GOOGLETEST_DIR} ${CMAKE_CURRENT_BINARY_DIR}/googletest)
    endif()
endif (WIN32 AND NOT MSVC)

# Check if we're being included from the top CMakeLists.txt
if (CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    # We're being built directly, so build core first
    add_subdirectory(${CORE_DIR} core)
    # Build recording library if enabled (core/src already adds it - guard against duplicate target)
    if (ENABLE_RECORDING AND NOT TARGET unrealng::recording)
        add_subdirectory(${ROOT_DIR}/core/recording recording)
    endif()
else()
    # I am called from other project with add_subdirectory()
    # So skip /src/core dependency build
endif()

# Compile all *.cpp files both from core/src and core/tests folders
file (GLOB_RECURSE TEST_CPP_FILES ${TESTS_DIR}/*.cpp)
EXCLUDE_FILES_FROM_DIR_IN_LIST("${TEST_CPP_FILES}" "/build/" FALSE OUTVAR TEST_CPP_FILES)      # Exclude /tests/build folder
EXCLUDE_FILES_FROM_DIR_IN_LIST("${TEST_CPP_FILES}" "/CMakeFiles/" FALSE OUTVAR TEST_CPP_FILES) # Exclude /CMakeFile autogenerated files

# Exclude recording tests if recording is disabled
if (NOT ENABLE_RECORDING)
    EXCLUDE_FILES_FROM_DIR_IN_LIST("${TEST_CPP_FILES}" "/recording/" FALSE OUTVAR TEST_CPP_FILES)
endif()

# MCP automation tests exercise the drogon-free MCP sources (dispatcher, tools,
# router, resources, target resolver), which depend on jsoncpp only. They are
# compiled into the test binary when the vendored jsoncpp_static target exists —
# i.e. when the unified build added core/automation (WebAPI + MCP enabled)
# before this directory. Otherwise the tests/automation folder is excluded.
if (TARGET jsoncpp_static)
    set(MCP_TESTS_AVAILABLE TRUE)
else()
    set(MCP_TESTS_AVAILABLE FALSE)
    EXCLUDE_FILES_FROM_DIR_IN_LIST("${TEST_CPP_FILES}" "/automation/" FALSE OUTVAR TEST_CPP_FILES)
endif()

set (SOURCES "${CORE_CPP_FILES};${TEST_CPP_FILES};${GOOGLETEST_FILES}")
message("TESTS:${TEST_CPP_FILES}")
message("SOURCES:${SOURCES}")

# Create test executable only if not already defined (prevents duplicates in unified builds)
if (NOT TARGET ${PROJECT_NAME})
    # Compile tests into executable
    add_executable(${PROJECT_NAME} ${SOURCES})

    # Declare _ALWAYS_USE_LOGGING for compiler
    target_compile_definitions(${PROJECT_NAME} PRIVATE _ALWAYS_USE_LOGGING)

    # Declare _DEBUG symbol for compiler
    if (CMAKE_BUILD_TYPE MATCHES Debug)
        message(STATUS "CMAKE_BUILD_TYPE:${CMAKE_BUILD_TYPE} - defining '_DEBUG'")
        target_compile_definitions(${PROJECT_NAME} PRIVATE _DEBUG)
    endif (CMAKE_BUILD_TYPE MATCHES Debug)

    # Enable test wrappers in code
    target_compile_definitions(${PROJECT_NAME} PRIVATE _CODE_UNDER_TEST)

    # MCP automation: compile the drogon-free MCP sources into the test binary.
    # Drogon appears only in automation-mcp.cpp / webapi-client.cpp, which are
    # deliberately NOT listed here; the sources below need jsoncpp only.
    if (MCP_TESTS_AVAILABLE)
        target_sources(${PROJECT_NAME} PRIVATE
            ${ROOT_DIR}/core/automation/mcp/src/mcp-dispatcher.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-tools.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-router.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-resources.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-symbols.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-analysis.cpp
            ${ROOT_DIR}/core/automation/mcp/src/mcp-media.cpp
            ${ROOT_DIR}/core/automation/mcp/src/target-resolver.cpp
        )
        target_include_directories(${PROJECT_NAME} PRIVATE ${ROOT_DIR}/core/automation/mcp/src)
        # Plain signature (no keyword) to match the existing target_link_libraries call below
        target_link_libraries(${PROJECT_NAME} jsoncpp_static)
    endif()

    # 'pch.h' is our precompiled header
    #target_precompile_headers(${PROJECT_NAME} PRIVATE ${TESTS_DIR}/pch.h)

    target_link_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/googletest)

    # All standard libraries required
    if (WIN32 AND MSVC)
        set(common_libraries gtest)
    endif (WIN32 AND MSVC)

    if (WIN32 AND MINGW)
        # Link all as static
        set_target_properties(${PROJECT_NAME} PROPERTIES LINK_FLAGS "-static-libgcc -static-libstdc++ -Wl,-Bstatic,--whole-archive -lwinpthread -liconv -Wl,--no-whole-archive")

        set(common_libraries shlwapi ws2_32)
    endif (WIN32 AND MINGW)

    if (UNIX AND NOT APPLE)
        set(CMAKE_CXX_FLAGS " -static-libstdc++")
        set(common_libraries -static-libstdc++ pthread gtest)
    endif (UNIX AND NOT APPLE)

    if (APPLE)
        target_link_directories(${PROJECT_NAME} PRIVATE /usr/lib)
        set(common_libraries pthread iconv gtest)
    endif (APPLE)

    target_link_libraries(${PROJECT_NAME}
        ${common_libraries}
        core
        $<$<BOOL:${ENABLE_RECORDING}>:recording>
    )

    # Enable CTest for Google Test discovery
    enable_testing()
    include(GoogleTest)
    gtest_discover_tests(${PROJECT_NAME})

    # Parallel test execution target
    # Uses GTest sharding for ~3x speedup (37s -> 12s on 4 cores)
    # Usage: cmake --build build --target test-parallel
    if(UNIX)
        add_custom_target(test-parallel
            COMMAND ${ROOT_DIR}/scripts/run-tests-parallel.sh $<TARGET_FILE:${PROJECT_NAME}>
            DEPENDS ${PROJECT_NAME}
            WORKING_DIRECTORY ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}
            COMMENT "Running tests with 4-way GTest sharding (~3x speedup)"
        )
    elseif(WIN32)
        add_custom_target(test-parallel
            COMMAND powershell -ExecutionPolicy Bypass -File ${ROOT_DIR}/scripts/run-tests-parallel.ps1 -Binary $<TARGET_FILE:${PROJECT_NAME}>
            DEPENDS ${PROJECT_NAME}
            WORKING_DIRECTORY ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}
            COMMENT "Running tests with 4-way GTest sharding (~3x speedup)"
        )
    endif()

    # Define data paths
    set(DATA_PATH "${ROOT_DIR}/data")

    # Copy resources using modern CMake syntax
    # Configs are deployed ONLY under configs/<model>/ - never as a flat unreal.ini
    add_custom_command(
        TARGET ${PROJECT_NAME} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${DATA_PATH}/configs" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/configs"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${DATA_PATH}/rom" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/rom"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${DATA_PATH}/testrom" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/rom"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${ROOT_DIR}/testdata" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/testdata"
        COMMENT "Copying runtime resource files to output directory"
    )
endif()
